cmake_minimum_required(VERSION 3.16)
project(prog5_climate VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PROG5_WITH_MQTT "Build libmosquitto MQTT support" OFF)
option(PROG5_WITH_UNO_Q "Build UNO Q arduino-router I2C adapter (msgpack-cxx)" OFF)
option(PROG5_SANITIZERS "Enable address/undefined sanitizers (GCC/Clang)" OFF)

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    if(PROG5_SANITIZERS)
        add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
        add_link_options(-fsanitize=address,undefined)
    endif()
endif()

add_library(sensor_core src/Hal.cpp src/Bmp280.cpp)
target_include_directories(sensor_core PUBLIC src)
add_library(stepper_core src/Stepper.cpp)
target_include_directories(stepper_core PUBLIC src)
add_library(matrix_text src/MatrixText.cpp)
target_include_directories(matrix_text PUBLIC src)

find_package(Threads REQUIRED)
add_library(station_support app/Publisher.cpp app/Sampler.cpp)
target_include_directories(station_support PUBLIC app)
target_link_libraries(station_support PUBLIC sensor_core Threads::Threads)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(station app/main.cpp)
    target_link_libraries(station PRIVATE station_support)
    if(PROG5_WITH_UNO_Q)
        find_path(MSGPACK_INCLUDE_DIR msgpack.hpp)
        if(NOT MSGPACK_INCLUDE_DIR)
            message(FATAL_ERROR "Install libmsgpack-cxx-dev for UNO Q Router RPC support")
        endif()
        add_library(uno_q_platform platform/uno_q/UnoQPlatform.cpp)
        target_include_directories(uno_q_platform PUBLIC platform/uno_q)
        target_include_directories(uno_q_platform SYSTEM PRIVATE ${MSGPACK_INCLUDE_DIR})
        target_compile_definitions(uno_q_platform PRIVATE MSGPACK_NO_BOOST)
        target_link_libraries(uno_q_platform PUBLIC sensor_core Threads::Threads)
        target_link_libraries(station PRIVATE uno_q_platform)
        target_compile_definitions(station PRIVATE PROG5_HAS_UNO_Q=1)
    endif()
    if(PROG5_WITH_MQTT)
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(MOSQUITTO REQUIRED IMPORTED_TARGET libmosquitto)
        add_library(mqtt_publisher app/MqttPublisher.cpp)
        target_include_directories(mqtt_publisher PUBLIC app)
        target_link_libraries(mqtt_publisher PUBLIC station_support PkgConfig::MOSQUITTO)
        target_link_libraries(station PRIVATE mqtt_publisher)
        target_compile_definitions(station PRIVATE PROG5_HAS_MQTT=1)
    endif()
elseif(PROG5_WITH_MQTT OR PROG5_WITH_UNO_Q)
    message(FATAL_ERROR "The UNO Q station application targets its onboard Debian Linux")
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(unit_tests tests/test_main.cpp tests/test_sensor.cpp
                              tests/test_stepper.cpp tests/test_app.cpp tests/test_matrix.cpp)
    target_link_libraries(unit_tests PRIVATE station_support stepper_core matrix_text)
    if(TARGET uno_q_platform)
        target_sources(unit_tests PRIVATE tests/test_router.cpp)
        target_link_libraries(unit_tests PRIVATE uno_q_platform)
        target_include_directories(unit_tests SYSTEM PRIVATE ${MSGPACK_INCLUDE_DIR})
        target_compile_definitions(unit_tests PRIVATE PROG5_TEST_ROUTER=1)
        target_compile_definitions(unit_tests PRIVATE MSGPACK_NO_BOOST)
    endif()
    add_test(NAME unit_tests COMMAND unit_tests)
    set_tests_properties(unit_tests PROPERTIES TIMEOUT 15)
    if(TARGET station)
        add_test(NAME station_simulation COMMAND station --fake --seconds 1 --period-ms 100)
        add_test(NAME invalid_address COMMAND station --fake --address 0x60 --seconds 1)
        set_tests_properties(invalid_address PROPERTIES WILL_FAIL TRUE)
    endif()
endif()
